cmake_minimum_required(VERSION 3.16)
project(parser C)

# wgvim: tree-sitter parsers are C11 (some scanners use static_assert). Under the deps'
# default C99 that compiled as a call to an undefined function, which only surfaced once the
# parsers were linked with -z now (see SECURITY-AUDIT.md).
set(CMAKE_C_STANDARD 11)

add_compile_options(-w)

file(GLOB source_files src/*.c)

add_library(parser
  MODULE
  ${source_files}
)
set_target_properties(
  parser
  PROPERTIES
    OUTPUT_NAME ${PARSERLANG}
    PREFIX ""
)

include_directories(src)

install(TARGETS parser LIBRARY DESTINATION lib/wgvim/parser)
